<script>
let cart = [];
let total = 0;

// ** YAHAN APNA WHATSAPP NUMBER DAALEIN (With Country Code e.g. 919876543210) **
const myPhoneNumber = "919876543210";

function addToCart(name, price) {
cart.push({ name, price });
total += price;
updateCart();
}

function updateCart() {
const cartItemsDiv = document.getElementById('cart-items');
const cartCountSpan = document.getElementById('cart-count');
const cartTotalSpan = document.getElementById('cart-total');

cartCountSpan.innerText = cart.length;
cartTotalSpan.innerText = total;

if (cart.length === 0) {
cartItemsDiv.innerHTML = '<p style="color:#777; margin-top:10px;">Cart khali hai.</p>';
return;
}

cartItemsDiv.innerHTML = '';
cart.forEach((item, index) => {
cartItemsDiv.innerHTML += `
<div class="cart-item">
<span>${item.name}</span>
<span>₹${item.price}</span>
</div>
`;
});
}

function sendWhatsAppOrder() {
if (cart.length === 0) {
alert("Pehle Cart me product add karein!");
return;
}

let message = "Namaste! Mujhe ye products order karne hain:\n\n";
cart.forEach((item, index) => {
message += `${index + 1}. ${item.name} - ₹${item.price}\n`;
});
message += `\nTotal Amount: ₹${total}\n\nKripya mera order confirm karein!`;

const encodedMessage = encodeURIComponent(message);
window.open(`https://wa.me/${myPhoneNumber="919432986924"}?text=${encodedMessage}`, '_blank');
}

function scrollToCart() {
document.getElementById('cart').scrollIntoView({ behavior: 'smooth' });
}
